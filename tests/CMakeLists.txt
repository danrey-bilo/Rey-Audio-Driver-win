cmake_minimum_required(VERSION 3.20)
project(ReyCapturePublicationContract LANGUAGES CXX)
# Independent of the private USB library and external ASIO SDK.
enable_testing()
add_executable(ReyCapturePublicationContract usb_capture_batch_contract.cpp)
target_compile_features(ReyCapturePublicationContract PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ReyCapturePublicationContract PRIVATE /W4 /WX)
else()
  target_compile_options(ReyCapturePublicationContract PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME completed_capture_before_render COMMAND ReyCapturePublicationContract)
add_executable(ReyEndpointContract endpoints_contract.cpp)
target_compile_features(ReyEndpointContract PRIVATE cxx_std_17)
if(MINGW)
  # Keep these standalone checks independent of the build toolchain's DLLs.
  target_link_options(ReyCapturePublicationContract PRIVATE -static)
  target_link_options(ReyEndpointContract PRIVATE -static)
endif()
if(MSVC)
  target_compile_options(ReyEndpointContract PRIVATE /W4 /WX)
else()
  target_compile_options(ReyEndpointContract PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
foreach(case pairs phases capture render clipping)
  add_test(NAME endpoint_${case} COMMAND ReyEndpointContract ${case})
endforeach()
