cmake_minimum_required(VERSION 3.20)
project(ReyAudioDriver VERSION 2.6.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 (AMD64) and a 64-bit toolchain")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]" OR CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
  message(FATAL_ERROR "Windows ARM64 is not supported")
endif()
include(cmake/AoIPDependency.cmake)
enable_language(RC)
option(PIAOIP_BUILD_ASIO "Build the legacy ASIO adapter and control panel" OFF)
option(PIAOIP_BUILD_SERVICE "Build the independent single-Pi Windows service" ON)
option(PIAOIP_ENABLE_USB "Build Pi5-AUSB transport support in the service" ON)
option(REY_BUILD_TESTS "Build service control contract tests" ON)
option(REY_BUILD_CONTROL "Build the WPF desktop and tray interface" ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(piaoip_windows_transport STATIC src/transport/iocp_receiver.cpp)
target_compile_features(piaoip_windows_transport PUBLIC cxx_std_17)
target_compile_definitions(piaoip_windows_transport PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(piaoip_windows_transport PUBLIC AoIP::core ws2_32)
if(PIAOIP_BUILD_SERVICE)
  add_library(piaoip_windows_engine STATIC src/engine/peer_session.cpp src/engine/session_engine.cpp
    src/platform/realtime.cpp src/config/settings.cpp src/control/discovery.cpp)
  target_compile_features(piaoip_windows_engine PUBLIC cxx_std_17)
  target_link_libraries(piaoip_windows_engine PUBLIC piaoip_windows_transport AoIP::core AoIP::peer avrt shell32 ole32 advapi32 iphlpapi uuid)
  add_executable(PiAoipService apps/aoip_service.cpp src/bridge/driver_bridge.cpp)
  set(PIAOIP_SETUPAPI_LIBRARY setupapi CACHE STRING "SetupAPI import library supplied by the Windows toolchain")
  target_link_libraries(PiAoipService PRIVATE piaoip_windows_engine "${PIAOIP_SETUPAPI_LIBRARY}")
  add_executable(ReyAudioService apps/rey_audio_service.cpp src/service/settings.cpp
    src/service/manager.cpp src/service/pipe_server.cpp src/bridge/driver_bridge.cpp
    src/platform/device_names.cpp)
  target_link_libraries(ReyAudioService PRIVATE piaoip_windows_engine "${PIAOIP_SETUPAPI_LIBRARY}")
  if(REY_BUILD_TESTS)
    enable_testing()
    add_executable(ReyServiceContractTests tests/service_contract.cpp src/service/settings.cpp)
    target_link_libraries(ReyServiceContractTests PRIVATE AoIP::core ws2_32)
    target_compile_features(ReyServiceContractTests PRIVATE cxx_std_17)
    if(MINGW)
      target_link_options(ReyServiceContractTests PRIVATE -static)
    endif()
    foreach(case defaults numbers profile routing isolation json peer)
      add_test(NAME rey_service_${case} COMMAND ReyServiceContractTests ${case})
      set_tests_properties(rey_service_${case} PROPERTIES TIMEOUT 10)
    endforeach()
  endif()
  if(PIAOIP_ENABLE_USB)
    include(cmake/AUSBDependency.cmake)
    target_sources(PiAoipService PRIVATE src/engine/usb_session.cpp)
    target_link_libraries(PiAoipService PRIVATE Pi5AUSB::windows)
    target_compile_definitions(PiAoipService PRIVATE PIAOIP_ENABLE_USB=1)
    target_sources(ReyAudioService PRIVATE src/engine/usb_session.cpp)
    target_link_libraries(ReyAudioService PRIVATE Pi5AUSB::windows)
    target_compile_definitions(ReyAudioService PRIVATE PIAOIP_ENABLE_USB=1)
  endif()
  if(MINGW)
    target_link_options(PiAoipService PRIVATE -static -municode)
    target_link_options(ReyAudioService PRIVATE -static -municode)
  endif()
  add_executable(PiAoipWasapiProbe apps/wasapi_probe.cpp)
  set_target_properties(PiAoipWasapiProbe PROPERTIES OUTPUT_NAME ReyAudioProbe)
  set(PIAOIP_WASAPI_HEADERS "" CACHE PATH "Optional missing WASAPI headers for a minimal MinGW toolchain")
  if(PIAOIP_WASAPI_HEADERS)
    target_include_directories(PiAoipWasapiProbe SYSTEM PRIVATE "${PIAOIP_WASAPI_HEADERS}")
  endif()
  target_compile_features(PiAoipWasapiProbe PRIVATE cxx_std_17)
  target_compile_definitions(PiAoipWasapiProbe PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(PiAoipWasapiProbe PRIVATE ole32 uuid avrt)
  if(MINGW)
    target_link_options(PiAoipWasapiProbe PRIVATE -static -municode)
  endif()
  foreach(target PiAoipService PiAoipWasapiProbe ReyAudioService)
    set(PIAOIP_RESOURCE_NAME "${target}")
    if(target STREQUAL "PiAoipService")
      set(PIAOIP_RESOURCE_DESCRIPTION "PiAoIP single-Pi network audio service")
    elseif(target STREQUAL "ReyAudioService")
      set(PIAOIP_RESOURCE_DESCRIPTION "Rey Audio Driver USB and AoIP service")
    else()
      set(PIAOIP_RESOURCE_DESCRIPTION "PiAoIP WASAPI endpoint diagnostic")
    endif()
    configure_file(src/platform/windows_version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc" @ONLY)
    target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc")
  endforeach()
  if(REY_BUILD_CONTROL)
    find_program(REY_POWERSHELL NAMES pwsh powershell REQUIRED)
    set(REY_CONTROL_OUTPUT "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ReyAudioControl.exe")
    add_custom_command(OUTPUT "${REY_CONTROL_OUTPUT}"
      COMMAND "${REY_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_control.ps1" -Output "${REY_CONTROL_OUTPUT}"
      DEPENDS tools/build_control.ps1 apps/ReyAudioControl/MainWindow.xaml
        apps/ReyAudioControl/Program.cs apps/ReyAudioControl/ViewModel.cs
        apps/ReyAudioControl/ServiceClient.cs apps/ReyAudioControl/app.manifest
        apps/ReyAudioControl/ReyAudio.ico VERBATIM)
    add_custom_target(ReyAudioControl ALL DEPENDS "${REY_CONTROL_OUTPUT}")
  endif()
endif()
if(PIAOIP_BUILD_ASIO)
enable_language(RC)
set(ASIO_SDK_DIR "$ENV{ASIO_SDK_DIR}" CACHE PATH "External Steinberg ASIO SDK root; not distributed here")
if(NOT EXISTS "${ASIO_SDK_DIR}/common/iasiodrv.h" OR NOT EXISTS "${ASIO_SDK_DIR}/common/asio.h")
  message(FATAL_ERROR "Set ASIO_SDK_DIR to a separately obtained Steinberg ASIO SDK. See docs/ASIO-SDK.md")
endif()
add_library(PiAoipAsio SHARED src/driver/asio_driver.cpp src/config/settings.cpp src/control/discovery.cpp src/ui/settings_panel.cpp src/platform/realtime.cpp src/ui/controls.rc src/driver/PiAoipAsio.def)
target_compile_features(PiAoipAsio PRIVATE cxx_std_17)
target_include_directories(PiAoipAsio PRIVATE "${ASIO_SDK_DIR}/common" src/ui)
target_link_libraries(PiAoipAsio PRIVATE piaoip_windows_transport AoIP::core ws2_32 ole32 advapi32 gdi32 iphlpapi uuid avrt comctl32 uxtheme shell32)
set_target_properties(PiAoipAsio PROPERTIES PREFIX "")
add_executable(PiAoipControl WIN32 apps/control_panel.cpp src/config/settings.cpp src/control/discovery.cpp src/ui/control_panel.rc)
target_include_directories(PiAoipControl PRIVATE "${ASIO_SDK_DIR}/common" src/ui)
target_link_libraries(PiAoipControl PRIVATE AoIP::core ole32 uuid ws2_32 advapi32 iphlpapi shell32 gdi32)
foreach(target PiAoipAsio PiAoipControl)
  target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_compile_features(${target} PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(${target} PRIVATE -static)
  endif()
endforeach()
endif()
