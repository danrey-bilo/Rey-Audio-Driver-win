cmake_minimum_required(VERSION 3.20)
project(ReyAudioDriver VERSION 2.8.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 (AMD64) and a 64-bit toolchain")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]" OR CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
  message(FATAL_ERROR "Windows ARM64 is not supported")
endif()
enable_language(RC)
add_compile_definitions(_WIN32_WINNT=0x0A00 WINVER=0x0A00)
option(REY_BUILD_TESTS "Build audio and service contract tests" ON)
option(REY_BUILD_CONTROL "Build the WPF desktop and tray interface" ON)
option(REY_BUILD_USB_ASIO "Build the USB ASIO frontend and diagnostic host" ON)
option(REY_ENABLE_TAG_BRIDGE "Enable the optional local Windows endpoint evaluation backend" OFF)
set(REY_SETUPAPI_LIBRARY setupapi CACHE STRING "SetupAPI import library")
set(REY_WASAPI_HEADERS "" CACHE PATH "Optional missing WASAPI headers for a minimal MinGW toolchain")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(cmake/AUSBDependency.cmake)

add_library(rey_asio_ipc STATIC src/asio/ipc_server.cpp src/asio/ipc_client.cpp)
target_compile_features(rey_asio_ipc PUBLIC cxx_std_17)
target_compile_definitions(rey_asio_ipc PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(rey_usb_engine STATIC src/engine/usb_session.cpp src/platform/realtime.cpp)
target_compile_features(rey_usb_engine PUBLIC cxx_std_17)
target_compile_definitions(rey_usb_engine PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(rey_usb_engine PUBLIC Pi5AUSB::windows avrt)
add_library(rey_service_core STATIC src/service/settings.cpp src/audio/mixer.cpp
  src/service/manager.cpp src/service/device_session.cpp src/service/commands.cpp
  src/service/asio_commands.cpp src/bridge/driver_bridge.cpp src/platform/device_names.cpp)
target_link_libraries(rey_service_core PUBLIC rey_usb_engine rey_asio_ipc "${REY_SETUPAPI_LIBRARY}" ole32 advapi32 uuid)
if(REY_ENABLE_TAG_BRIDGE)
  target_sources(rey_service_core PRIVATE src/bridge/gateway_bridge.cpp)
  target_compile_definitions(rey_service_core PRIVATE REY_ENABLE_TAG_BRIDGE=1)
endif()
add_executable(ReyAudioService apps/rey_audio_service.cpp src/service/pipe_server.cpp)
target_link_libraries(ReyAudioService PRIVATE rey_service_core)
add_executable(ReyAudioProbe apps/wasapi_probe.cpp)
target_compile_features(ReyAudioProbe PRIVATE cxx_std_17)
target_compile_definitions(ReyAudioProbe PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(ReyAudioProbe PRIVATE ole32 uuid avrt)
if(REY_WASAPI_HEADERS)
  target_include_directories(ReyAudioProbe SYSTEM PRIVATE "${REY_WASAPI_HEADERS}")
endif()
if(MINGW)
  target_link_options(ReyAudioService PRIVATE -static -municode)
  target_link_options(ReyAudioProbe PRIVATE -static -municode)
endif()

if(REY_BUILD_USB_ASIO)
  set(ASIO_SDK_DIR "$ENV{ASIO_SDK_DIR}" CACHE PATH "External Steinberg ASIO SDK root; not distributed here")
  if(NOT EXISTS "${ASIO_SDK_DIR}/common/iasiodrv.h" OR NOT EXISTS "${ASIO_SDK_DIR}/common/asio.h")
    message(FATAL_ERROR "Set ASIO_SDK_DIR to a separately obtained Steinberg ASIO SDK. See docs/ASIO-SDK.md")
  endif()
  add_library(ReyAudioAsio SHARED src/asio/usb_driver.cpp src/platform/realtime.cpp src/asio/ReyAudioAsio.def)
  target_include_directories(ReyAudioAsio PRIVATE "${ASIO_SDK_DIR}/common")
  target_link_libraries(ReyAudioAsio PRIVATE rey_asio_ipc ole32 uuid advapi32 shell32 avrt)
  set_target_properties(ReyAudioAsio PROPERTIES PREFIX "")
  add_executable(ReyAsioProbe apps/usb_asio_probe.cpp src/platform/realtime.cpp)
  target_include_directories(ReyAsioProbe PRIVATE "${ASIO_SDK_DIR}/common")
  target_compile_features(ReyAsioProbe PRIVATE cxx_std_17)
  target_compile_definitions(ReyAsioProbe PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ReyAsioProbe PRIVATE ole32 uuid avrt)
  if(MINGW)
    target_link_options(ReyAudioAsio PRIVATE -static)
    target_link_options(ReyAsioProbe PRIVATE -static -municode)
  endif()
endif()
set(REY_RESOURCE_TARGETS ReyAudioService ReyAudioProbe)
if(REY_BUILD_USB_ASIO)
  list(APPEND REY_RESOURCE_TARGETS ReyAudioAsio ReyAsioProbe)
endif()
foreach(target IN LISTS REY_RESOURCE_TARGETS)
  set(REY_RESOURCE_NAME "${target}")
  set(REY_RESOURCE_DESCRIPTION "Rey Audio Driver USB ${target}")
  set(REY_RESOURCE_TYPE VFT_APP)
  set(REY_RESOURCE_FILENAME "${target}.exe")
  if(target STREQUAL "ReyAudioAsio")
    set(REY_RESOURCE_TYPE VFT_DLL)
    set(REY_RESOURCE_FILENAME "${target}.dll")
  endif()
  configure_file(src/platform/windows_version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc" @ONLY)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc")
endforeach()
if(REY_BUILD_CONTROL)
  find_program(REY_POWERSHELL NAMES pwsh powershell REQUIRED)
  set(REY_CONTROL_OUTPUT "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ReyAudioControl.exe")
  add_custom_command(OUTPUT "${REY_CONTROL_OUTPUT}"
    COMMAND "${REY_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_control.ps1" -Output "${REY_CONTROL_OUTPUT}"
    DEPENDS tools/build_control.ps1 apps/ReyAudioControl/MainWindow.xaml
      apps/ReyAudioControl/Program.cs apps/ReyAudioControl/ViewModel.cs
      apps/ReyAudioControl/ServiceClient.cs apps/ReyAudioControl/app.manifest
      apps/ReyAudioControl/MixerChannel.cs apps/ReyAudioControl/AsioPreferences.cs
      apps/ReyAudioControl/ReyAudio.ico VERBATIM)
  add_custom_target(ReyAudioControl ALL DEPENDS "${REY_CONTROL_OUTPUT}")
endif()
if(REY_BUILD_TESTS)
  enable_testing()
  add_executable(ReyAsioIpcTests tests/asio_ipc_contract.cpp)
  target_link_libraries(ReyAsioIpcTests PRIVATE rey_asio_ipc)
  add_executable(ReyServiceContractTests tests/service_contract.cpp src/service/settings.cpp src/service/commands.cpp src/audio/mixer.cpp)
  target_link_libraries(ReyServiceContractTests PRIVATE advapi32)
  target_compile_features(ReyServiceContractTests PRIVATE cxx_std_17)
  add_executable(ReyMixerContractTests tests/mixer_contract.cpp src/audio/mixer.cpp)
  target_compile_features(ReyMixerContractTests PRIVATE cxx_std_17)
  add_executable(ReyEndpointContractTests tests/endpoints_contract.cpp)
  target_compile_features(ReyEndpointContractTests PRIVATE cxx_std_17)
  add_executable(ReyDeviceContractTests tests/devices_contract.cpp)
  target_link_libraries(ReyDeviceContractTests PRIVATE rey_service_core)
  foreach(target ReyAsioIpcTests ReyServiceContractTests ReyMixerContractTests ReyEndpointContractTests ReyDeviceContractTests)
    if(MINGW)
      target_link_options(${target} PRIVATE -static)
    endif()
  endforeach()
  foreach(case invalid ownership capture render overflow)
    add_test(NAME rey_asio_ipc_${case} COMMAND ReyAsioIpcTests ${case})
  endforeach()
  foreach(case defaults numbers profile json persistence corruption legacy_v1 legacy_v2 commands alignment)
    add_test(NAME rey_service_${case} COMMAND ReyServiceContractTests ${case})
  endforeach()
  foreach(case unity solo_mute gain_clip ramp bounds)
    add_test(NAME rey_mixer_${case} COMMAND ReyMixerContractTests ${case})
  endforeach()
  foreach(case pairs capture render clipping)
    add_test(NAME rey_endpoints_${case} COMMAND ReyEndpointContractTests ${case})
  endforeach()
  foreach(case empty single sticky ambiguous disconnect persistence retired asio_auth)
    add_test(NAME rey_device_${case} COMMAND ReyDeviceContractTests ${case})
  endforeach()
  get_property(REY_TESTS DIRECTORY PROPERTY TESTS)
  set_tests_properties(${REY_TESTS} PROPERTIES TIMEOUT 15)
endif()
